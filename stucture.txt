event-qr-system/
├── docker-compose.yml       # [จำเป็น] ตั้งค่า Multi-container Docker และ Timezone (Asia/Bangkok)
├── Dockerfile               # [จำเป็น] สร้าง Docker Image สำหรับ Backend (Node.js)
├── init.sql                 # [จำเป็น] สคริปต์ SQL สร้างฐานข้อมูลเริ่มต้นอัตโนมัติ
├── package.json             # [จำเป็น] จัดการไลบรารีและแพ็กเกจ Node.js
├── server.js                # [จำเป็น] Backend หลักจัดการ API ทั้งหมด
└── public/                  # โฟลเดอร์เก็บไฟล์หน้าเว็บฝั่ง Frontend
    ├── admin.html           # [จำเป็น] หน้าจอผู้ดูแลระบบ (จัดการ Event, Import Excel, ออกรายงาน)
    ├── login.html           # [จำเป็น] หน้าจอเข้าสู่ระบบตรวจสอบสิทธิ์
    ├── register.html        # [จำเป็น] หน้าจอลงทะเบียนผู้เข้าร่วมงาน / ส่ง QR Code ทางอีเมล
    ├── success.html         # [ใหม่/จำเป็น] หน้าจอแสดงผลหลังลงทะเบียนสำเร็จ พร้อมรายละเอียดและปุ่มกลับไปลงทะเบียนซ้ำ
    ├── scanner.html         # [จำเป็น] หน้าจอสำหรับสแกน QR Code เช็คอินเข้างาน
    └── js/                  # โฟลเดอร์สคริปต์ฝั่ง Frontend
        ├── admin.js         # [จำเป็น] ควบคุมหน้า admin.html
        └── login.js         # [จำเป็น] ควบคุมหน้า login.html
